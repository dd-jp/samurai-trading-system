"""Attribution gate: reproduce the PUBLISHED arms under the OLD labelling, then
show what the corrected labelling moves. Same bars, same engine, same grid."""
import importlib.util, json, os, re
from datetime import datetime

os.environ["SAMURAI_DATA_DIR"] = "/tmp/s685"
os.environ["TF"] = "1Min"
P = "/Users/ddjp/Documents/projects/samurai-trading-system/.claude/worktrees/agent-a039547045b0467cf/docs/research/18-threshold-study.py"
spec = importlib.util.spec_from_file_location("m", P)
m = importlib.util.module_from_spec(spec)
spec.loader.exec_module(m)

LEV, COST = 3.0, 0.41
TPS, SLS = [2.0, 4.0, 6.0], [1.5, 3.0]
sessions = m.load_sessions("TSLA", "1Min")
days = sorted(sessions)
have = set(days)
print("sessions %d  %s..%s" % (len(days), days[0], days[-1]))


def fit(dayset):
    rows = []
    for tp in TPS:
        for sl in SLS:
            s = m.stats(m.simulate(sessions, dayset, LEV, tp, sl, COST))
            if s:
                rows.append((tp, sl, s))
    rows.sort(key=lambda r: -r[2]["exp"])
    return rows[0] if rows else None


def show(tag, s):
    if s is None:
        print("   %-22s none" % tag)
        return
    print("   %-22s exp %+.4f%%  n=%-4d SE %.3f  t %+.2f" % (tag, s["exp"], s["n"], s["se"], s["t"]))


# ---------- OLD labelling, verbatim from the superseded revision -------------
name, quarter, eps_old = "tesla", re.compile(r"\bq[1-4]\b"), re.compile(r"\beps\b")
old_marks = set()
for line in open("/tmp/s685/news_TSLA.jsonl"):
    n = json.loads(line)
    h = n["h"].lower()
    if not (h.startswith(name) and quarter.search(h) and eps_old.search(h)):
        continue
    t = datetime.fromisoformat(n["t"].replace("Z", "+00:00")).astimezone(m.ET)
    old_marks.add((t.date(), "next" if (t.hour * 60 + t.minute) >= 16 * 60 else "same"))
old_react = set()
for d, when in old_marks:
    if when == "same":
        if d in have:
            old_react.add(d)
    else:
        for cand in days:
            if cand > d:
                old_react.add(cand)
                break

new_react, excluded = m.reaction_dates("TSLA", have)

print("\n== REACTION-DAY RECONCILIATION")
print("   OLD %d (ADR-0018 published 46)   NEW %d   excluded intraday %d"
      % (len(old_react & have), len(new_react & have), len(excluded & have)))
print("   in OLD, not in NEW:", sorted((old_react & have) - (new_react & have)))
print("   in NEW, not in OLD:", sorted((new_react & have) - (old_react & have)))

for tag, react, excl in (("OLD", old_react, set()), ("NEW", new_react, excluded)):
    split = [d for d in days if d.year <= 2022]
    oos = [d for d in days if d.year >= 2023]
    ev_is = [d for d in split if d in react and d not in excl]
    ord_is = [d for d in split if d not in react and d not in excl]
    ev_oos = [d for d in oos if d in react and d not in excl]
    ord_oos = [d for d in oos if d not in react and d not in excl]
    b_pool, b_ord, b_ev = fit(split), fit(ord_is), (fit(ev_is) if len(ev_is) >= 8 else None)
    print("\n== %s LABELLING   events IS=%d OOS=%d" % (tag, len(ev_is), len(ev_oos)))
    print("   frozen levels: pooled TP%+.1f/SL-%.1f  ordinary TP%+.1f/SL-%.1f  event TP%+.1f/SL-%.1f"
          % (b_pool[0], b_pool[1], b_ord[0], b_ord[1],
             b_ev[0] if b_ev else 0, b_ev[1] if b_ev else 0))
    show("grid pooled", m.stats(m.simulate(sessions, oos, LEV, b_pool[0], b_pool[1], COST)))
    s_ord = m.stats(m.simulate(sessions, ord_oos, LEV, b_ord[0], b_ord[1], COST))
    show("ordinary only", s_ord)
    s_ev = m.stats(m.simulate(sessions, ev_oos, LEV, b_ev[0], b_ev[1], COST)) if b_ev else None
    show("event only", s_ev)
    if s_ev and s_ord:
        d_ = s_ev["exp"] - s_ord["exp"]
        se = (s_ev["se"] ** 2 + s_ord["se"] ** 2) ** 0.5
        print("   %-22s %+.4f%%  SE %.3f  t %+.2f" % ("event minus ordinary", d_, se, d_ / se))
    combo = (m.simulate(sessions, ord_oos, LEV, b_ord[0], b_ord[1], COST)
             + (m.simulate(sessions, ev_oos, LEV, b_ev[0], b_ev[1], COST) if b_ev else []))
    show("combination", m.stats(combo))
